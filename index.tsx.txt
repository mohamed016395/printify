function BestSellers() {
  const { data: products } = useSuspenseQuery(productsQuery());
  const list = products.slice(0, 10);
  return (
    <section className="mx-auto max-w-7xl px-4 py-16 sm:px-6">
      <SectionHead
        eyebrow="02 — Best sellers"
        title="Crowd favorites"
        action={
          <Button variant="glass" asChild>
            <Link to="/shop">
              View all <ArrowRight className="h-4 w-4" />
            </Link>
          </Button>
        }
      />
      {list.length === 0 ? (
        <div className="rounded-2xl border border-dashed border-border bg-surface/40 p-12 text-center text-muted-foreground">
          No products found yet — new prints are on the way.
        </div>
      ) : (
        <Carousel opts={{ align: "start", loop: list.length > 3 }} className="relative">
          <CarouselContent className="-ml-4">
            {list.map((p) => (
              <CarouselItem key={p.id} className="basis-[78%] pl-4 sm:basis-1/2 lg:basis-1/4">
                <ProductCard product={p} className="h-full" />
              </CarouselItem>
            ))}
          </CarouselContent>
          <div className="mt-6 flex justify-end gap-2">
            <CarouselPrevious className="static translate-y-0 border-border bg-surface" />
            <CarouselNext className="static translate-y-0 border-border bg-surface" />
          </div>
        </Carousel>
      )}
    </section>
  );
}
